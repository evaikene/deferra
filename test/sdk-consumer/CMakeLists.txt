cmake_minimum_required(VERSION 3.20)
project(JobUSDKConsumer LANGUAGES CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
enable_testing()

if(SDK_CASE STREQUAL "default")
    find_package(JobU CONFIG REQUIRED)
elseif(SDK_CASE STREQUAL "discovery")
    find_package(JobU 0.1.0 EXACT CONFIG REQUIRED COMPONENTS Core)
    find_package(JobU CONFIG REQUIRED COMPONENTS Client CLI)
    find_package(JobU CONFIG REQUIRED COMPONENTS Core)
elseif(SDK_CASE STREQUAL "unknown")
    find_package(JobU CONFIG REQUIRED COMPONENTS Unknown)
elseif(SDK_CASE STREQUAL "version")
    find_package(JobU 0.2.0 EXACT CONFIG REQUIRED COMPONENTS Core)
elseif(SDK_CASE STREQUAL "missing-sqlite")
    find_package(JobU CONFIG REQUIRED COMPONENTS SQLite)
elseif(SDK_CASE STREQUAL "optional")
    find_package(JobU CONFIG REQUIRED COMPONENTS Core OPTIONAL_COMPONENTS SQLite Unknown)
    if(JobU_SQLite_FOUND OR JobU_Unknown_FOUND OR TARGET JobU::db-sqlite)
        message(FATAL_ERROR "Unavailable optional components were reported as available")
    endif()
elseif(SDK_CASE STREQUAL "optional-http")
    find_package(JobU CONFIG REQUIRED COMPONENTS Core OPTIONAL_COMPONENTS HTTP)
    if(JobU_HTTP_FOUND OR TARGET JobU::net-http)
        message(FATAL_ERROR "HTTP was imported without its external dependency")
    endif()
else()
    set(component_Core Core)
    set(component_Shared Core)
    set(component_Client Client)
    set(component_SQLite SQLite)
    set(component_HTTP HTTP)
    set(component_CLI CLI)
    if(NOT DEFINED component_${SDK_CASE})
        message(FATAL_ERROR "Unknown SDK fixture: ${SDK_CASE}")
    endif()
    find_package(JobU CONFIG REQUIRED COMPONENTS ${component_${SDK_CASE}})
endif()

# No JobU target may import a producer checkout, build tree, or original prefix
# after relocation. Inspect usage requirements as well as archive locations.
string(FIND "${JobU_DIR}/" "${SDK_PREFIX}/" prefix_position)
if(NOT prefix_position EQUAL 0)
    message(FATAL_ERROR "JobU was discovered outside the selected installed prefix: ${JobU_DIR}")
endif()
set(property_report "JobU_DIR=${JobU_DIR}\n")
foreach(target core net db rpc jobu jobu-client db-sqlite jobu-sqlite net-http jobu-http jobu-cli)
    if(NOT TARGET JobU::${target})
        continue()
    endif()
    get_target_property(imported JobU::${target} IMPORTED)
    if(NOT imported)
        message(FATAL_ERROR "Consumer unexpectedly sees a build-tree target: ${target}")
    endif()
    set(properties INTERFACE_INCLUDE_DIRECTORIES INTERFACE_SYSTEM_INCLUDE_DIRECTORIES
        INTERFACE_LINK_LIBRARIES INTERFACE_SOURCES INTERFACE_COMPILE_FEATURES
        INTERFACE_COMPILE_OPTIONS INTERFACE_COMPILE_DEFINITIONS INTERFACE_LINK_OPTIONS IMPORTED_LOCATION)
    get_target_property(configurations JobU::${target} IMPORTED_CONFIGURATIONS)
    foreach(configuration IN LISTS configurations)
        list(APPEND properties "IMPORTED_LOCATION_${configuration}")
    endforeach()
    foreach(property IN LISTS properties)
        get_target_property(value JobU::${target} "${property}")
        foreach(forbidden IN LISTS SDK_FORBIDDEN_PATHS)
            string(FIND "${value}" "${forbidden}" position)
            if(NOT position EQUAL -1)
                message(FATAL_ERROR "${target}.${property} leaks ${forbidden}: ${value}")
            endif()
        endforeach()
        if(value MATCHES "nlohmann_json|Catch2")
            message(FATAL_ERROR "${target}.${property} leaks a producer-only dependency: ${value}")
        endif()
        string(APPEND property_report "${target}.${property}=${value}\n")
    endforeach()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/target-properties.txt" "${property_report}")

if(SDK_CASE STREQUAL "Shared")
    add_library(core-shared SHARED core.cpp)
    target_link_libraries(core-shared PRIVATE JobU::core)
    add_executable(consumer core_main.cpp)
    target_link_libraries(consumer PRIVATE core-shared)
elseif(SDK_CASE STREQUAL "Client" OR SDK_CASE STREQUAL "discovery")
    add_executable(consumer client.cpp)
    target_link_libraries(consumer PRIVATE JobU::jobu-client)
elseif(SDK_CASE STREQUAL "SQLite")
    add_executable(consumer sqlite.cpp)
    target_link_libraries(consumer PRIVATE JobU::jobu-sqlite)
elseif(SDK_CASE STREQUAL "HTTP")
    add_executable(consumer http.cpp)
    target_link_libraries(consumer PRIVATE JobU::jobu-http JobU::net-http)
elseif(SDK_CASE STREQUAL "CLI")
    add_executable(consumer cli.cpp)
    target_link_libraries(consumer PRIVATE JobU::jobu-cli)
else()
    add_executable(consumer core.cpp core_main.cpp)
    target_link_libraries(consumer PRIVATE JobU::core)
endif()
add_test(NAME consumer COMMAND consumer)
