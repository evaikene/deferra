cmake_minimum_required(VERSION 3.20)
project(PhpCxxConsumer LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(PHPCXX_SOURCE_DIR)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${PHPCXX_SOURCE_DIR}" php-cxx)
else()
    find_package(PhpCxx 0.1.0 EXACT CONFIG REQUIRED)
endif()
get_target_property(PHP_BINARY PhpCxx::PHP PHPCXX_PHP_BINARY)

if(NOT TARGET PhpCxx::PhpCxx)
    message(FATAL_ERROR "Public target PhpCxx::PhpCxx is missing")
endif()
get_target_property(library_type PhpCxx::PhpCxx TYPE)
get_target_property(dependencies PhpCxx::PhpCxx INTERFACE_LINK_LIBRARIES)
if(NOT library_type STREQUAL "STATIC_LIBRARY" OR NOT dependencies STREQUAL "PhpCxx::PHP")
    message(FATAL_ERROR "php-cxx must be an independent static library: ${library_type}; ${dependencies}")
endif()

if(PHPCXX_CONSUMER_COMPILE_ONLY)
    add_library(consumer OBJECT build.cpp module.cpp functions.cpp objects.cpp)
    target_link_libraries(consumer PRIVATE PhpCxx::PhpCxx)
else()
    add_executable(consumer build.cpp)
    target_link_libraries(consumer PRIVATE PhpCxx::PhpCxx)
    add_library(consumer-module MODULE module.cpp functions.cpp objects.cpp)
    target_link_libraries(consumer-module PRIVATE PhpCxx::PhpCxx)
    set_target_properties(consumer-module PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
    if(APPLE)
        target_link_options(consumer-module PRIVATE "LINKER:-undefined,dynamic_lookup")
    endif()
    enable_testing()
    add_test(NAME consumer COMMAND consumer)
    add_test(NAME module COMMAND "${PHP_BINARY}" -n
        "-dextension=$<TARGET_FILE:consumer-module>" -r
        "if (!extension_loaded('phpcxx_consumer')) { exit(1); }")
    add_test(NAME conversions COMMAND "${PHP_BINARY}" -n
        "-dextension=$<TARGET_FILE:consumer-module>" "${CMAKE_CURRENT_SOURCE_DIR}/check.php")
    add_test(NAME objects COMMAND "${PHP_BINARY}" -n
        "-dextension=$<TARGET_FILE:consumer-module>" "${CMAKE_CURRENT_SOURCE_DIR}/objects-check.php")
endif()
